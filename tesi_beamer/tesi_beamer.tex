\begin{frame}{Il problema della stabilità nei sistemi dinamici}
  \begin{itemize}
    \item \blu{Origine fisica e celeste:} Lo studio della stabilità a lungo termine ha radici storiche secolari nella meccanica celeste newtoniana (problema degli $n$-corpi e stabilità del Sistema Solare).
    \medskip
    \item \blu{Il teorema di Arnold (1963):}
    \begin{itemize}
      \item Sistema planetario: Sole e $n \ge 2$ pianeti con masse molto più piccole.
      \item Trascurando mutue interazioni, ogni pianeta descrive un'ellisse kepleriana (sistema \alert{completamente integrabile}).
      \item Per interazioni sufficientemente deboli, nello spazio delle fasi esiste un insieme di \textbf{misura di Lebesgue strettamente positiva} di condizioni iniziali (posizione, velocità) che generano moti \alert{quasi-periodici}.
    \end{itemize}
    \medskip
    \item \blu{La dimostrazione rigorosa di Herman (anni '90):}
    \begin{itemize}
      \item Arnold diede una dimostrazione completa solo per il problema piano dei 3 corpi.
      \item Herman formulò un \textbf{teorema astratto di tori invarianti}, da cui dedusse in modo rigoroso il teorema di Arnold nella sua completezza.
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Soluzioni quasi-periodiche}
    In generale, dato $\alpha = (\alpha_1, \dots, \alpha_n) \in \R^n$ tale che $k \cdot \alpha \neq 0$ per ogni $k \in \Z^n \setminus \{0\}$, una soluzione $u$ associata ad una condizione iniziale nello spazio delle fasi $\mathcal{P}$ è detta \textit{quasi-periodica} di frequenza $\alpha$ se esiste un embedding del toro
    \[
      \theta \mapsto U(\theta)
    \]
    tale che $u(t) = U(\alpha t)$ per ogni $t$ nel dominio massimale di definizione della soluzione. \\ 
    In seguito lavoreremo sempre con moti quasi-periodici.
\end{frame}

% slide 3
\begin{frame}{Sistemi completamente integrabili}
  Consideriamo lo spazio delle fasi $T^*\T^n \cong \T^n \times \R^n = \{(\theta, r)\}$ dotato della forma simplettica standard:
  \[
    \omega = \sum_{j=1}^n d\theta_j \wedge dr_j, \qquad X_H = \begin{pmatrix} \partial_r H \\ -\partial_\theta H \end{pmatrix}.
  \]
  e consideriamo una hamiltoniana $H(\theta, r) = H_0(r)$:
    \[
      \begin{cases}
        \dot{\theta} = \partial_r H_0(r) =: \alpha(r) \\
        \dot{r} = 0
      \end{cases}
      \implies 
      \begin{cases}
        \theta(t) = \theta_0 + t\,\alpha(r_0) \pmod{2\pi\Z^n} \\
        r(t) = r_0
      \end{cases}
    \]
  \begin{itemize}
    \item Lo spazio delle fasi è \textbf{foliato da tori invarianti lagrangiani} $\T^n \times \{r_0\}$.
    \item Il flusso su ciascun toro è una traslazione lineare a frequenza $\alpha = \alpha(r_0)$.
  \end{itemize}
   \alert{Domanda:} il moto caratteristico dei sistemi completamente integrabili persiste per perturbazioni sufficientemente piccole?
\end{frame}

% slide 4
\begin{frame}{Sistemi quasi-integrabili e teoria perturbativa classica}
  Consideriamo un sistema perturbato, dipendente formalmente da un parametro $0 \le \eps \ll 1$:
  \[
    H(\theta, r) = H_0(r) + \eps H_1(\theta, r) + \eps^2 H_2(\theta, r) + O(\varepsilon^3)
  \]
  \pause
  \blu{Metodo classico:} cercare un simplettomorfismo generato dal flusso al tempo $1$ di un'hamiltoniana ausiliaria $F$:
  \[
    H \circ \phi_1 = H_0 + \eps \big(H_1 + \{H_0, F\}\big) + O(\eps^2) = H_0 + \eps \big(H_1 - \alpha \cdot \partial_\theta F\big) + O(\eps^2)
  \]
  \pause
  Scomponendo $H_1 = \overline{H}_1(r) + \widetilde{H}_1(\theta, r)$ con $\int_{\T^n} \widetilde{H}_1 d\theta = 0$, si ottiene l'\alert{equazione omologica}:
  \[
    \mathcal{L}_\alpha F = \alpha \cdot \partial_\theta F = \widetilde{H}_1(\theta, r) \implies F(\theta) = \sum_{k \in \Z^n \setminus \{0\}} \frac{c_k}{i(k \cdot \alpha)} e^{i k \cdot \theta}
  \]
 \[
    \inf_{k \neq 0}|k \cdot \alpha| = 0
 \]
\end{frame}

% slide 5
\begin{frame}{Condizione aritmetica diofantea}
  Siccome $\inf_{k \neq 0} |k \cdot \alpha| = 0$ è senza speranza cercare di ottenere una stima del tipo $|k \cdot \alpha| \geq \delta$, per qualche $\delta > 0$.
  \begin{block}{Vettori $(\gamma, \tau)$-diofantei}
    Un vettore di frequenza $\alpha \in \R^n$ è detto \blu{$(\gamma, \tau)$-diofanteo} ($D_{\gamma, \tau}$) se:
    \[
      |k \cdot \alpha| \ge \frac{\gamma}{|k|^\tau}, \qquad \forall k \in \Z^n \setminus \{0\} \quad (\gamma > 0, \; \tau > n - 1)
    \]
  \end{block}
  \pause
  \begin{itemize}
    \item \blu{Quantità vettori diofantei:} \underline{quasi tutti} i vettori di $\R^n$ sono diofantei, $m(D_{\gamma, \tau}) = 1 - O(\gamma)$, possiamo visualizzarlo facilmente in $\R^2$. 
    \item \blu{Risolubilità dell'equazione omologica:} Per ogni $g$ analitica su $\T^n_{s+\sigma}$ con $\int g = 0$, esiste un'unica soluzione $f$ analitica su $\T^n_s$ soddisfacente (stima di Cauchy):
    \[
      f = \mathcal{L}_\alpha^{-1}g \qquad |f|_s \le C \gamma^{-1} \sigma^{-(n+\tau)} |g|_{s+\sigma}
    \]
  \end{itemize}
  \alert{Prezzo da pagare:} inevitabile \textbf{perdita di analiticità} $\sigma > 0$ (ma arbitraria $\rightsquigarrow$ scale di spazi di Banach).
\end{frame}

% slide 7
\begin{frame}{Il teorema di Kolmogorov e la superconvergenza}
  \begin{block}{Teorema dei tori invarianti (Kolmogorov, 1954)}
    Fissato $\alpha \in D_{\gamma, \tau}$ e una hamiltoniana che possiede un toro invariante $\alpha$-quasi-periodico $\T^n \times \{0\}$
    \[
      K^0 = c + \alpha \cdot r + \frac{1}{2}K^0_2(\theta)r \cdot r + O(r^3), \qquad \det \int K^0_2(\theta) \, d\theta \neq 0
    \]
    per ogni perturbazione $H$ sufficientemente vicina a $K^0$, esiste una trasformazione simplettica $G$ vicina all'identità e $K$ vicina a $K^0$, tale che $H = K \circ G$, a meno di traslare le azioni per $H$.
  \end{block}
  \pause
  \blu{L'idea rivoluzionaria di Kolmogorov:}
  \begin{itemize}
    \item Usare uno \textbf{schema iterativo quadratico alla Newton}:
    \[
      \eps_{m+1} \sim \eps_m^2 \qquad (\text{o più precisamente } \eps_m^{\chi}, \; 1 < \chi < 2)
    \]
    \item La \alert{superconvergenza} quadratica schiaccia la perdita geometrica di analiticità $\sigma_m^{-(n+\tau)}$, garantendo la convergenza della successione di trasformazioni simplettiche.
  \end{itemize}
\end{frame}

% slide 8
\begin{frame}{La strategia di Herman: separazione delle difficoltà}
  Negli anni '90, Herman riformula la persistenza dei tori invarianti isolando due passaggi distinti:
  \medskip
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{\blu{Passo 1: Analisi Dura (Hard Analysis)}}
        \begin{itemize}
          \item Fissiamo la frequenza $\alpha \in D_{\gamma, \tau}$.
          \item \textbf{Nessuna ipotesi di non degenerazione} su $K^0$.
          \item Si introduce un controtermine $\beta \cdot r$ con \alert{$\beta \in \R^n$} (da determinare):
          \[
            H - \beta \cdot r = K \circ G
          \]
          \item Inversione locale dell'operatore non lineare $\phi : (K, G, \beta) \mapsto K \circ G + \beta \cdot r$, $|H - K^0|_{s + \sigma} \ll 1$.
        \end{itemize}
      \end{block}
    \end{column}
    \pause
    \begin{column}{0.48\textwidth}
      \begin{block}{\blu{Passo 2: Eliminazione del Parametro}}
        \begin{itemize}
          \item Il controtermine $\beta$ rompe la coniugazione dinamica di $H$.
          \pause
          \item Sotto l'ipotesi $\det \int K^0_2 \neq 0$, considero $r \mapsto r + R$ (parametri liberi):
          \[
            \beta = \beta(R), \qquad \det \frac{\partial \beta}{\partial R}(0) \neq 0
          \]
          \item \textbf{Teorema classico della funzione inversa} per annullare $\beta$:
          \[
            \beta(R^*) = 0 \implies H_{R^*} = K \circ G
          \]
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% slide 9
\begin{frame}{Il Teorema \textit{Twist} di Herman}
  Definiamo la classe delle hamiltoniane con toro invariante $\alpha$-quasi-periodico all'origine:
  \[
    \mathcal{K}^\alpha = \left\{ K \in \mathcal{H} : K(\theta, r) = c + \alpha \cdot r + O(r^2) \right\} \implies X_K|_{T^n_0} = \begin{pmatrix} \alpha \\ 0 \end{pmatrix}.
  \]
  \begin{block}{Teorema (Twisted Conjugacy di Herman)}
    Siano $\alpha \in D_{\gamma, \tau}$ e $K^0 \in \mathcal{K}^\alpha$. Per ogni hamiltoniana $H$ sufficientemente vicina a $K^0$ nella topologia analitica, esistono unici:
    \begin{enumerate}
      \item $K \in \mathcal{K}^\alpha$ vicino a $K^0$,
      \item $G$ simplettomorfismo esatto vicino all'identità,
      \item un vettore di controtermine $\beta \in \R^n$ vicino a $0$,
    \end{enumerate}
    tali che:
    \[
      H - \beta \cdot r = K \circ G
    \]
    in un intorno del toro $T_0^n = \T^n \times \{0\}$.
  \end{block}
\end{frame}

% slide 10
\begin{frame}{Necessità geometrica del controtermine $\beta$}
  \begin{itemize}
    \item \blu{Invarianza del vettore frequenza:} se $\dot{\theta} = \alpha$ è coniugato a $\dot{\theta}' = \beta$ tramite un diffeomorfismo $\varphi(\theta) = \theta + v(\theta)$, necessariamente $\alpha = \beta$. \\
    In realtà, già Arnold si accorse di questo studiando le rotazioni del cerchio $\T$ e fu lui ad avere l'idea di separare il problema in due passaggi, poi ripresa da Herman.
  \end{itemize}
  \bigbreak  
  \begin{itemize}
    \item \blu{Non stabilità aperta:} avere un toro invariante con frequenza fissata $\alpha$ \textbf{non è una proprietà aperta} (cioè stabile per piccole perturbazioni).
  \end{itemize}
  \pause
  \begin{block}{Hamiltoniana 'modificata'}
    Osserviamo che $G^{-1}(T^n_0)$ è invariante per il flusso della hamiltoniana
    \[
      \widehat{H} := H - \beta \cdot r = K \circ G
    \]
  \end{block}
\end{frame}

% slide 11
\begin{frame}{Struttura dei simplettomorfismi ammissibili}
  Per preservare la natura lagrangiana del toro, consideriamo lo spazio $\mathcal{G}$ dei germi di simplettomorfismi esatti $G: (\theta, r) \mapsto (\Theta, R)$:
  \[
    G(\theta, r) = \Big(\varphi(\theta), \; {}^t\varphi'(\theta)^{-1}\big(r + \rho(\theta)\big)\Big)
  \]
  \begin{itemize}
    \item $\varphi : \T^n \to \T^n$ è un diffeomorfismo del toro vicino all'identità che raddrizza il moto sul toro ($\varphi(0) = 0$).
    \item $\rho(\theta) = dS(\theta)$ è una $1$-forma \textbf{esatta} su $\T^n$ che trasla le azioni per riposizionare il toro nello spazio delle fasi.
  \end{itemize}
\end{frame}

% slide 12
\begin{frame}{Setting funzionale: scale di Banach e l'operatore $\phi$}
  Per gestire rigorosamente le perdite di analiticità senza restringere brutalmente i domini:
  \begin{itemize}
    \item Dominio complesso: $T^n_s = \{(\theta, r) \in \C^n/2\pi\Z^n \times \C^n : \max(|\mathrm{Im}\,\theta|, |r|) \le s\}$.
    \item Spazio di Banach $(\mathcal{H}_s, |\cdot|_s)$ delle funzioni analitiche con norma del sup su $T^n_s$.
  \end{itemize}
A questo punto è ben definito l'operatore $\phi$ da invertire:
    \[
      \phi : (K, G, \beta) \in \mathcal{K}^\alpha_{s+\sigma} \times \mathcal{G}_{s+\sigma}^\sigma \times \R^n \longmapsto H = K \circ G + \beta \cdot r \in \mathcal{H}_s
    \]
\end{frame}

% slide 13
\begin{frame}{Costruzione della successione con metodo alla Newton}
  Applichiamo il metodo alla Newton per risolvere l'equazione fortemente non lineare $\phi(K, G, \beta) = H$, quindi determinare $x := (K, G, \beta)$ avendo noto $y := H$. In particolare, vogliamo trovare lo zero della funzione $F : x \mapsto \phi(x) - y$. Sia
  \[
    f : x \mapsto x + \phi'(x)^{-1}(y - \phi(x))
  \]
  e poniamo
  \[
    x_0 := (K^0, \operatorname{id}, 0) \qquad x_n := f(f(\dots f(x_0)))
  \]
  Partendo da $x_0$, ad ogni passo risolviamo l'equazione linearizzata nel punto dato dal passo precedente.
  \begin{alertblock}{Stime fondamentali alla Nash-Moser}
    \[
      |\phi'(x)^{-1}\delta H|_s \le C' \sigma^{-(\tau+n+1)} |\delta H|_{G, s+\sigma}, \qquad |\phi''(x)\delta x^{\otimes 2}|_{G, s} \le C'' \sigma^{-2} |\delta x|_{s+\sigma}^2
    \]
  \end{alertblock}
\end{frame}

% slide 14
\begin{frame}{Perdita controllata di analiticità}
  Fissiamo le riduzioni di dominio ad ogni passo dello schema di Newton:
  \[
    \sigma_k = \frac{\sigma}{6} 2^{-k}, \qquad s_{k+1} = s_k - 3\sigma_k, \qquad 3\sum_{k=0}^\infty \sigma_k = \sigma > 0
  \]
  \pause
  \begin{itemize}
      \item \blu{Residuo quadratico}:
      \[
        |x_{k+1} - x_k|_{s_{k+1}} \le c_k |x_k - x_{k-1}|_{s_k}^2
      \]
      con costante di perdita $c_k = \frac{1}{2} C \sigma_k^{-\tau} \sim 2^{k\tau}$
      \medskip
      \item \blu{Superconvergenza:}
      \[
        |x_{k+1} - x_k|_{s_{k+1}} \le \Big(\eps \prod_{j=0}^k c_j^{2^{-j}}\Big)^{2^k} \le \text{cost} \cdot \eps^{2^k}
      \]
  \end{itemize}
\end{frame}

% slide 15
\begin{frame}{Ritorno al Teorema di Kolmogorov e Conclusioni}
  \blu{Passo finale: annullamento del controtermine $\beta$}
  \begin{itemize}
    \item Sia $Q = \frac{1}{2}\int K^0_2$ e introduciamo una famiglia di perturbazioni per traslazione delle azioni: $r \mapsto r + R$.
    \item Per l'hamiltoniana traslata imperturbata $K^0(r+R)$, il controtermine vale:
    \[
      \widehat{\beta}^0(R) = 2QR + O(R^2) \implies \left.\frac{\partial \widehat{\beta}^0}{\partial R}\right|_{R=0} = 2Q
    \]
    \item Poiché per ipotesi di Kolmogorov $\det Q \neq 0$ (non degenerazione), $R \mapsto \beta(R)$ è un \textbf{diffeomorfismo locale} per ogni $H$ sufficientemente vicina a $K^0$.
    \item Esiste un unico $R^*$ tale che $\beta(R^*) = 0$:
    \[
      H(\theta, r+R^*) = K \circ G(\theta, r)
    \]
    Il toro invariante con moto $\alpha$-quasi-periodico persiste.
  \end{itemize}
\end{frame}
