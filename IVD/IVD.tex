\chapter{Varietà}
\section{Varietà topologiche}
\begin{defn}
    uno spazio topologico $M$ è detto \co{varietà topologica} se soddisfa le seguenti: \\
    \gr{1.} $M$ è di Hausdorff \\
    \gr{2.} per ogni $p \in M$ esistono $U \subset M$ intorno aperto di $p$ e $\vp : U \mapsto \vp(U) \subset \R^n$ omeomorfismo, ovvero ogni punto possiede un intorno omeomorfo a $\R^n$, per qualche $n \in \mathbb{N}$. \\
    \gr{3.} $M$ soddisfa il secondo assioma di numerabilità, ovvero possiede una base numerabile $\B$ della topologia. \\ \\
    Una coppia $(U, \vp)$ come nel punto $(2)$ è detta \co{carta locale} di $M$. Ogni carta induce delle \co{coordinate locali} su $M$, infatti, dato $p \in M$, si ha
    \begin{align}
        \vp(p) = (x_1, \dots, x_n) = (x_1(p), \dots, x_n(p)) \in \R^n.
    \end{align}
    Quindi chiamiamo coordinate locali le mappe $x_i : U \longrightarrow \R$, $i = 1, \dots, n$ definite da
    \begin{align}
        x_i : p \in U \mapsto (\vp(p), e_i)
    \end{align}
    dove con $(e_i)_{1 \leq i \leq n}$ identifichiamo la base canonica di $\R^n$ e con $(\cdot, \cdot)$ il prodotto scalare euclideo in $\R^n$. \\ \\
    Mentre $\vp^{-1} : \vp(U) \longrightarrow U$ è detta \co{parametrizzazione locale} di $M$.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $M$ una varietà topologica, due carte locali $(U, \vp)$ e $(V, \psi)$ di $M$ sono dette $k$-\co{compatibili} se vale una delle seguenti: \\
    \gr{1.} $U \cap V = \emptyset$. \\
    \gr{2.} $U \cap V \neq \emptyset$ e $\psi \circ \vp^{-1} : \vp(U \cap V) \longrightarrow \psi(U \cap V)$ è un $\C^k$-diffeomorfismo.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $M$ una varietà topologica, chiamiamo \co{atlante} $\C^k$-\co{differenziabile} una famiglia di carte locale $\A = \{(U_\alpha, \vp_\alpha)\}_{\alpha \in A}$ che sono a due a due $\C^k$-compatibili.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $\A_1, \A_2$ atlanti $\C^k$-differenziabili di $M$. $\A_1, \A_2$ sono detti \co{equivalenti} se la loro unione è ancora un atlante $\C^k$-differenziabile di $M$. \\
    Si può dimostrare che questa relazione è una relazione di equivalenza. 
\end{defn}
\begin{defn}
    una \co{struttura} $\C^k$-\co{differenziabile} su $M$ è una classe di equivalenza di atlanti $\C^k$-differenziabili di $M$. \\
    Fissata una struttura $\C^k$-differenziabile su $M$, chiamiamo $M$ \co{varietà} $\C^k$-differenziabile.
\end{defn}
Da ora in poi, $k \geq 1$ sarà sempre un intero e quando ometteremo la classe di regolarità della struttura differenziabile intendiamo che la struttura è $\C^\infty$-differenziabile, ovvero $\C^k$-differenziabile per ogni $k \geq 1$.
\begin{rmk}
    l'unione di tutti gli atlanti in una classe di equivalenza è detto \co{atlante massimale}, esiste per il \co{lemma di Zorn}.
\end{rmk}
\begin{ex}
    \gr{1.} $M = \R$ è varietà differenzabile con atlante canonico $(\R, \id_\R)$\\
    \gr{2.} $(\R, f)$ dove $f : x \in \R \mapsto x^3 \in \R$ \\ \\
    Osserviamo che gli atlanti $(\R, \id)$ e $(\R, f)$ non sono compatibili. Infatti, $\R \cap \R = \R$, tuttavia
    \begin{align}
        f \circ \id^{-1} = f
    \end{align}
    è un omeomorfismo ma la funzione inversa
    \begin{align}
        f^{-1} : y \in \R \mapsto y^{1/3}
    \end{align}
    non è differenziabile in $y = 0$. Pertanto $f \circ \id^{-1} = f$ non è un diffeomorfismo.
\end{ex}
\begin{ex}
    sia $n \geq 2$ e consideriamo la sfera unitaria $S^{n - 1} := \{x \in \R^n : \|x\| = 1\}$ e poniamo
    \begin{align}
        p_N = (0, \dots, 0, 1) \qquad p_S = (1, 0, \dots, 0)
    \end{align}
    e definiamo gli aperti
    \begin{align}
        U_N = S^{n - 1} \setminus \{p_N\} \qquad U_S := S^{n - 1} \setminus \{p_S\}
    \end{align}
    Osserviamo immediatamente che $\{U_N, U_S\}$ è un ricoprimento aperto di $S^{n - 1}$. \\
    Ora vogliamo definire le mappe $\vp_N, \vp_S$ da associare agli aperti $U_N, U_S$. In particolare, immaginiamo di dividere la sfera all'equatore con un iperpiano e di tracciare una retta che parte da $p_N$ e interseca un punto qualsiasi $x$ della sfera fino a raggiungere anche l'iperpiano. Vogliamo definire $\vp_N$ come la mappa che ad ogni punto $x$ associa il punto dell'iperpiano che la retta passante per $p_N$ e $x$ interseca. La retta $r$ passante per $p_N$ e $x = (x_1, \dots, x_n)$ è data da
    \begin{align}
        r = \{(1 - t)p_N + tx : t \in \mathbb{R}\} = \{(x_1, \dots, 1 - t(1 - x_n)) \in \R^n : t \in \R\}
    \end{align}
    mentre l'iperpiano che passa per l'equatore è
    \begin{align}
        E = \{x = (x_1, \dots, x_n) \in \R^n : x_n = 0\}
    \end{align}
    Notiamo che per $t^* = \frac{1}{1 - x_n}$ si ha $(1 - t^*)p_N + t^*x \in E$, pertanto definiamo $\vp_N$ come la mappa
    \begin{align}
        \vp_N : x = (x_1, \dots, x_n) \in U_N \mapsto \frac{1}{1 - x_n}(x_1, \dots, x_{n - 1}) \in \R^{n - 1}
    \end{align}
    Verifichiamo che $\vp_N$ sia un omeomorfismo. Si vede immediatamente che $\vp_N$ è iniettiva, infatti
    \begin{align}
        \frac{1}{1 - x_n}(x_1, \dots, x_{n - 1}) = \frac{1}{1 - y_n}(y_1, \dots, y_{n - 1})
    \end{align}
    se e solo se $x_i = y_i$ per ogni $i = 1, \dots, n$. Inoltre, dato $z \in \R^{n - 1}$, esiste $\wtz = (\wtz_1, \dots, \wtz_n) \in S^{n - 1}$ tale che $\varphi_N(\wtz) = z$, infatti, affinché
    \begin{align}
        \frac{1}{1 - \wtz_n}(\wtz_1, \dots, \wtz_{n - 1}) = (z_1, \dots, z_{n - 1})
    \end{align}
    basta scegliere $\wtz_i = z_i/\lambda$ per $i = 1, \dots, n - 1$ per qualche opportuno $\lambda > 0$ e
    \begin{align}
        \wtz_n = 1 - \frac{1}{\lambda}
    \end{align}
    In particolare $\wtz_n$ deve soddisfare
    \begin{align}
        \wtz_1^2 + \dots + \wtz_n^2 = 1
    \end{align}
    ovvero
    \begin{align}
        \frac{1}{\lambda^2}\|z\|^2 + 1 + \frac{1}{\lambda^2} - \frac{2}{\lambda} = 1
    \end{align}
    che equivale a
    \begin{align}
        \frac{1}{\lambda}\left(\frac{\|z\|^2 + 1}{\lambda} - 2\right) = 0
    \end{align}
    Quindi si ottiene
    \begin{align}
        \lambda = \frac{\|z\|^2 + 1}{2}
    \end{align}
    Pertanto $\vp_N$ suriettiva. Infine, si vede immediatamente che $\vp_N$ è continua su $U_N$ (dato che $p_N \not\in U_N)$, quindi $\vp_N$ omeomorfismo. \\ 
    Analogamente si costruisce $\vp_S$, in particolare si ottiene
    \begin{align}
        \vp_S : x = (x_1, \dots, x_n) \in S^n \mapsto \frac{1}{1 + x_n}(x_1, \dots, x_{n - 1}) \in \R^{n - 1}
    \end{align}
    Verifichiamo ora che $\vp_N \circ \vp_S^{-1} : \vp_S(U_S) \longrightarrow \vp_N(U_N)$ è di classe $\C^\infty$, ovvero che le carte locali $(U_N, \vp_N)$ e $(U_S, \vp_S)$ sono $\C^\infty$-compatibili.
\end{ex}
