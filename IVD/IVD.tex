\chapter{Varietà}
\begin{defn}
    uno spazio topologico $M$ è detto \co{varietà topologica} se soddisfa le seguenti: \\
    \gr{1.} $M$ è di Hausdorff \\
    \gr{2.} per ogni $p \in M$ esistono $U \subset M$ intorno aperto di $p$ e $\vp : U \mapsto \vp(U) \subset \R^n$ omeomorfismo, ovvero ogni punto possiede un intorno omeomorfo a $\R^n$, per qualche $n \in \mathbb{N}$. \\
    \gr{3.} $M$ soddisfa il secondo assioma di numerabilità, ovvero possiede una base numerabile $\B$ della topologia. \\ \\
    Una coppia $(U, \vp)$ come nel punto $(2)$ è detta \co{carta locale} di $M$. Ogni carta induce delle \co{coordinate locali} su $M$, infatti, dato $p \in M$, si ha
    \begin{align}
        \vp(p) = (x_1, \dots, x_n) = (x_1(p), \dots, x_n(p)) \in \R^n.
    \end{align}
    Quindi chiamiamo coordinate locali le mappe $x_i : U \longrightarrow \R$, $i = 1, \dots, n$ definite da
    \begin{align}
        x_i : p \in U \mapsto (\vp(p), e_i)
    \end{align}
    dove con $(e_i)_{1 \leq i \leq n}$ identifichiamo la base canonica di $\R^n$ e con $(\cdot, \cdot)$ il prodotto scalare euclideo in $\R^n$. \\ \\
    Mentre $\vp^{-1} : \vp(U) \longrightarrow U$ è detta \co{parametrizzazione locale} di $M$.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $M$ una varietà topologica, due carte locali $(U, \vp)$ e $(V, \psi)$ di $M$ sono dette $k$-\co{compatibili} se vale una delle seguenti: \\
    \gr{1.} $U \cap V = \emptyset$. \\
    \gr{2.} $U \cap V \neq \emptyset$ e $\psi \circ \vp^{-1} : \vp(U \cap V) \longrightarrow \psi(U \cap V)$ è un $\C^k$-diffeomorfismo.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $M$ una varietà topologica, chiamiamo \co{atlante} $\C^k$-\co{differenziabile} una famiglia di carte locale $\A = \{(U_\alpha, \vp_\alpha)\}_{\alpha \in A}$ che sono a due a due $\C^k$-compatibili.
\end{defn}
\begin{defn}
    siano $k \geq 1$ e $\A_1, \A_2$ atlanti $\C^k$-differenziabili di $M$. $\A_1, \A_2$ sono detti \co{equivalenti} se la loro unione è ancora un atlante $\C^k$-differenziabile di $M$. \\
    Si può dimostrare che questa relazione è una relazione di equivalenza. 
\end{defn}
\begin{defn}
    una \co{struttura} $\C^k$-\co{differenziabile} su $M$ è una classe di equivalenza di atlanti $\C^k$-differenziabili di $M$. \\
    Fissata una struttura $\C^k$-differenziabile su $M$, chiamiamo $M$ \co{varietà} $\C^k$-differenziabile.
\end{defn}
Da ora in poi, $k \geq 1$ sarà sempre un intero e quando ometteremo la classe di regolarità della struttura differenziabile intendiamo che la struttura è $\C^\infty$-differenziabile, ovvero $\C^k$-differenziabile per ogni $k \geq 1$.
\begin{rmk}
    l'unione di tutti gli atlanti in una classe di equivalenza è detto \co{atlante massimale}, esiste per il \co{lemma di Zorn}.
\end{rmk}
\begin{ex}
    \gr{1.} $M = \R$ è varietà differenzabile con atlante canonico $(\R, \id_\R)$\\
    \gr{2.} $(\R, f)$ dove $f : x \in \R \mapsto x^3 \in \R$ \\ \\
    Osserviamo che gli atlanti $(\R, \id)$ e $(\R, f)$ non sono compatibili. Infatti, $\R \cap \R = \R$, tuttavia
    \begin{align}
        f \circ \id^{-1} = f
    \end{align}
    è un omeomorfismo ma la funzione inversa
    \begin{align}
        f^{-1} : y \in \R \mapsto y^{1/3}
    \end{align}
    non è differenziabile in $y = 0$.
\end{ex}
\begin{ex}
    sia $n \geq 2$ e consideriamo la sfera unitaria $S^{n - 1} := \{x \in \R^n : \|x\| = 1\}$ e poniamo
    \begin{align}
        p_N = (0, \dots, 0, 1) \qquad p_S = (1, 0, \dots, 0)
    \end{align}
    e definiamo gli aperti
    \begin{align}
        U_N = S^{n - 1} \setminus \{p_N\} \qquad U_S := S^{n - 1} \setminus \{p_S\}
    \end{align}
    Osserviamo immediatamente che $\{U_N, U_S\}$ è un ricoprimento aperto di $S^{n - 1}$. \\
    Ora vogliamo definire le mappe $\vp_N, \vp_S$ da associare agli aperti $U_N, U_S$. In particolare, immaginiamo di dividere la sfera all'equatore con un iperpiano e di tracciare una retta che parte da $p_N$ e interseca un punto qualsiasi $x$ della sfera fino a raggiungere anche l'iperpiano. Vogliamo definire $\vp_N$ come la mappa che ad ogni punto $x$ associa il punto dell'iperpiano che la retta passante per $p_N$ e $x$ interseca. La retta $r$ passante per $p_N$ e $x = (x_1, \dots, x_n)$ è data da
    \begin{align}
        r = \{(1 - t)p_N + tx : t \in \mathbb{R}\} = \{(tx_1, \dots, 1 - t(1 - x_n)) \in \R^n : t \in \R\}
    \end{align}
    mentre l'iperpiano che passa per l'equatore è
    \begin{align}
        E = \{x = (x_1, \dots, x_n) \in \R^n : x_n = 0\} \cong \R^{n - 1}
    \end{align}
    Notiamo che per $t^* = \frac{1}{1 - x_n}$ si ha $(1 - t^*)p_N + t^*x \in E$, pertanto definiamo $\vp_N$ come la mappa
    \begin{align}
        \vp_N : x = (x_1, \dots, x_n) \in U_N \mapsto \frac{1}{1 - x_n}(x_1, \dots, x_{n - 1}) \in \R^{n - 1}
    \end{align}
    Verifichiamo che $\vp_N$ sia un omeomorfismo. Si vede immediatamente che $\vp_N$ è iniettiva, infatti
    \begin{align}
        \frac{1}{1 - x_n}(x_1, \dots, x_{n - 1}) = \frac{1}{1 - y_n}(y_1, \dots, y_{n - 1})
    \end{align}
    se e solo se $x_i = y_i$ per ogni $i = 1, \dots, n$. Inoltre, dato $z \in \R^{n - 1}$, esiste $\wtz = (\wtz_1, \dots, \wtz_n) \in S^{n - 1}$ tale che $\varphi_N(\wtz) = z$, infatti, affinché
    \begin{align}
        \frac{1}{1 - \wtz_n}(\wtz_1, \dots, \wtz_{n - 1}) = (z_1, \dots, z_{n - 1})
    \end{align}
    basta scegliere $\wtz_i = z_i/\lambda$ per $i = 1, \dots, n - 1$ per qualche opportuno $\lambda > 0$ e
    \begin{align}
        \wtz_n = 1 - \frac{1}{\lambda}
    \end{align}
    In particolare $\wtz_n$ deve soddisfare
    \begin{align}
        \wtz_1^2 + \dots + \wtz_n^2 = 1
    \end{align}
    ovvero
    \begin{align}
        \frac{1}{\lambda^2}\|z\|^2 + 1 + \frac{1}{\lambda^2} - \frac{2}{\lambda} = 1
    \end{align}
    che equivale a
    \begin{align}
        \frac{1}{\lambda}\left(\frac{\|z\|^2 + 1}{\lambda} - 2\right) = 0
    \end{align}
    Quindi si ottiene
    \begin{align}
        \lambda = \frac{\|z\|^2 + 1}{2}
    \end{align}
    Pertanto $\vp_N$ suriettiva. \\
    L'inversa di $\vp_N$ è data da
    \begin{align}
        \vp_N^{-1} : x = (x_1, \dots, x_{n - 1}) \in \R^{n - 1} \mapsto \left(x_1, \dots, x_{n - 1}, \frac{\|x\|^2 - 1}{\|x\|^2 + 1}\right)
    \end{align}
    Si vede immediatamente che $\vp_N^{-1}$ è biettiva e continua. \\
    Infine, si vede immediatamente che $\vp_N$ è continua su $U_N$ (dato che $p_N \not\in U_N)$, quindi $\vp_N$ omeomorfismo. \\ 
    Analogamente si costruisce $\vp_S$, in particolare si ottiene
    \begin{align}
        \vp_S : x = (x_1, \dots, x_n) \in S^n \mapsto \frac{1}{1 + x_n}(x_1, \dots, x_{n - 1}) \in \R^{n - 1}
    \end{align}
    Verifichiamo ora che le carte locali $(U_N, \vp_N)$ e $(U_S, \vp_S)$ sono $\C^\infty$-compatibili, ovvero che $\vp_S \circ \vp_N^{-1} : \vp_N(U_S \cap U_N) \longrightarrow \vp_S(U_N \cap U_S)$ è di classe $\C^\infty$. Dato $x \vp_N(\in U_N \cap U_S)$ si ha che
    \begin{align}
        \vp_S \circ \vp_N^{-1}(x) = \frac{2\|x\|^2}{\|x\|^2 + 1}(x_1, \dots, x_{n - 1})
    \end{align}
    quindi si vede immeditamente che $\vp_S \circ \vp_N^{-1}$ è $\C^\infty$.
\end{ex}

\begin{defn}\label{C^k(M, R)}
    sia $M$ una varietà $\C^k$-differenziabile, dato $p \in M$ diciamo che $f : M \longrightarrow \R$ è $\C^k$ in un intorno di $p$ se esiste una carta $(U, \vp)$ di $M$ tale che $p \in U$ e
    \begin{align}
        f \circ \vp^{-1} : \vp(U) \longrightarrow \R
    \end{align} 
    è $\C^k$ in un intorno di $\vp(p)$. \\ \\
    Se $f$ è $\C^k$ in un intorno di $p$ per ogni $k \geq 0$ diciamo che $f$ è \co{liscia}. Inoltre, se $f$ è $\C^k$ ($\C^\infty$) in un intorno di ogni $p \in M$ diciamo che $f$ è $\C^k$ ($\C^\infty$). Poniamo
    \begin{align}
        \C^k(M) := \{f : M \longrightarrow \R : f \text{ è } \C^k\}
    \end{align}
\end{defn}
\begin{prop}\label{ChartIndipC^k(M, R)}
    la definizione \ref{C^k(M, R)} non dipende dalla scelta della carta.
\end{prop}
\begin{proof}
    siano $p \in M$, $f$ $\C^k$ in un intorno di $p$ e $(U, \vp)$ e $(V, \psi)$ due carte $\C^k$-compatibili di $M$ con $p \in U \cap V$, ovvero tali che $\vp \circ \psi^{-1}$ è $\C^k$ su $\psi(U \cap V)$. \\
    Supponiamo (senza perdita di generalità) che $f \circ \vp^{-1}$ è $\C^k$ in un intorno di $\vp(p)$, allora
    \begin{align}
        f \circ \vp^{-1} \circ (\vp \circ \psi^{-1}) = f \circ \psi^{-1}
    \end{align}
    è $\C^k$ in un intorno di $\psi(p)$ contenuto in $\psi(U \cap V)$ poiché composizione di mappe $\C^k$ nel medesimo intorno.
\end{proof}
\begin{defn}\label{C^k(M, N)}
    siano $M, N$ varietà $\C^k$-differenziabili e $f : M \longrightarrow N$ continua. Diciamo che $f$ è $\C^k$ in un intorno di $p \in M$ se esistono $(U, \vp)$ carta locale di $M$ tale che $p \in U$ e $(V, \psi)$ carta locale di $N$ tale che $f(p) \in V$ tali che
    \begin{align}
        \psi \circ f \circ \vp^{-1} : \vp(U \cap f^{-1}(V)) \longrightarrow \psi(V \cap f(U))
    \end{align}
    è $\C^k$ in un intorno di $\vp(p)$.
\end{defn}
\begin{rmk}
    l'ipotesi $f$ continua serve a garantire che $f^{-1}(V)$ sia un aperto.
\end{rmk}
\begin{prop}
    la definizione \ref{C^k(M, N)} non dipende dalla scelta delle carte.
\end{prop}
\begin{proof}
    la dimostrazione è analoga a quella della proposizione \ref{ChartIndipC^k(M, R)}.
\end{proof}
\begin{lemma}
    siano $f : M \longrightarrow N$ e $g : N \longrightarrow L$ mappe $\C^k$ tra varietà, allora $g \circ f : M \longrightarrow L$ è $\C^k$.
\end{lemma}
\begin{proof}
    fissato $p \in M$, per ipotesi esistono $(U, \vp), (V \psi)$ e $(W, \eta)$ carte locali di $M, N$ e $L$ rispettivamente tali che $p \in U$, $f(p) \in V$ e $g(f(p)) \in W$ e
    \begin{align}
        \psi \circ f \circ \vp^{-1} \qquad \eta \circ g \circ \psi^{-1}
    \end{align}
    sono $\C^k$ in un intorno di $\vp(p)$ e in un intorno di $\psi(f(p))$ rispettivamente. Allora
    \begin{align}
        (\eta \circ g \circ \psi^{-1}) \circ (\psi \circ f \circ \vp^{-1}) = \eta \circ (g \circ f) \circ \vp^{-1}
    \end{align}
    è $\C^k$ in un intorno di $\vp(p)$.
\end{proof}
\begin{defn}
    siano $M, N$ varietà $\C^k$-differenziabili, $f : M \longrightarrow N$ è detto $\C^k$-\co{diffeomorifsmo} se è $\C^k$, invertibile e con inversa $\C^k$. \\
    Se esiste tale $f$ da $M$ in $N$ diciamo che $M$ e $N$ sono diffeomorfi e scriviamo $M \cong N$.
\end{defn}
\begin{defn}
    siano $M, N$ varietà $\C^k$-differenziabili, $f : M \longrightarrow N$ è detto $\C^k$-\co{diffeomorfismo locale} se per ogni $p \in M$ esiste $U \subset M$ intorno aperto di $p$ tale che $f|_U : U \longrightarrow f(U)$ è un $\C^k$-diffeomorfismo. \\ \\
    Ovviamente notiamo che ogni diffeomorfismo è anche un diffeomorfismo locale.
\end{defn}
\begin{rmk}
    data $f : X \longrightarrow Y$ omeomorfismo locale tra spazi topologici, si ha che $f$ è una mappa aperta. Infatti, dato $U \subset X$ aperto, per ipotesi, per ogni $x \in U$ esiste $U_x \subset X$ intorno aperto di $x$ tale che $f|_{U_x} : U_x \longrightarrow f(U_x)$ è un omeomorfismo. Inoltre, dato che $U$ è un aperto, per ogni $x \in U$ esiste $V_x \subset U$ intorno aperto di $x$. \\
    Concludiamo che
    \begin{align}
        f(U) = \bigcup_{x \in U} f|_{U_x}(U_x \cap V_x)
    \end{align}
    unione di una famiglia di aperti, pertanto $f(U)$ aperto, quindi $f$ mappa aperta.
\end{rmk}

\chapter{Gruppi di Lie}
\begin{defn}
    un \co{gruppo di Lie} è una varietà differenziabile che ammette struttura di gruppo tale che le operazioni di prodotto e inversione sono lisce.
\end{defn}
\begin{rmk}
    dato $G$ gruppo di Lie, è sufficiente verificare che la mappa $j : (a, b) \mapsto a \cdot b^{-1}$ è liscia.
\end{rmk}
\begin{ex}
    $GL_n(\R)$ ammette struttura di gruppo di Lie, infatti per definizione
    \begin{align}
        GL_n(\R) = \{A \in M_n(\R) : \det A \neq 0\}
    \end{align}
    e $\det : M_n(\R) \cong \R^{n^2} \longrightarrow \R$ è continua, dato che è essenzialmente una funzione polinomiale. Inoltre, la moltiplicazione tra matrici è liscia, dato he è quadratica in ogni entrata, e anche l'inversione è liscia in $GL_n(\R)$ dato che $\det A \neq 0$ se $A \in GL_n(\R)$.
\end{ex}
\begin{prop}
    se $G, H$ sono gruppi di Lie, allora $G \times H$ ammette struttura di gruppo di Lie.
\end{prop}
\begin{proof}
    se $G, H$ sono gruppi, anche $G \times H$ ammette struttura di gruppo (è sufficiente definire le operazione singolarmente per le due componenti $(g, h)$). Inoltre, sappiamo che il prodotto cartesiano di due varietà $\C^k$-differenziabili ammette struttura di varietà $\C^k$-differenziabile (è sufficiente prendere il prodotto cartesiano delle carte e definire coppie di omeomorfismi locali).
\end{proof}
\begin{cor}
    $(S^1)^n = \T^n$ è un gruppo di Lie.
\end{cor}
\begin{defn}
    dati $G, H$ gruppi di Lie, $f : G \longrightarrow H$ è detto \co{omomorfismo di gruppi di Lie} se è un omomorfismo di gruppi ed in più è liscia. \\ \\
    Diciamo che $f$ è un \co{isomorfismo di gruppi di Lie} se anche l'inversa è un omomorfismo di gruppi di Lie.
\end{defn}
\begin{ex}
    $\det : GL_n(\R) \longrightarrow (\R^*, \cdot)$ è un omeomorfismo di gruppi, infatti $\det AB = \det A \cdot \det B$.
\end{ex}

\begin{defn}
    sia $X$ uno spazio topologico connesso e localmente connesso per archi, la coppia $(\wtX, \pi)$, dove $\wtX$ è uno spazio topologico connesso e $\pi : \wtX \longrightarrow X$, è detta \co{rivestimento topologico} se $\pi$ è suriettiva e per ogni $x \in X$ esiste $U \subset X$ intorno aperto di $x$, detto \co{intorno ben ricoperto}, tale che $\pi^{-1}(U)$ è unione disgiunta di aperti connessi $(V_i)_{i \in I}$ di $\wtX$ e la restrizione $\pi_{V_i} : V_i \longrightarrow U$ è un omeomorfismo per ogni $i \in I$. \\ \\
    $(\wtX, \pi)$ è detto \co{rivestimento universale} se $\wtX$ è semplicemente connesso.
\end{defn}
\begin{defn}
    sia $M$ una varietà differenziabile connessa, un rivestimento $(\wtM, \pi)$ è detto \co{rivestimento liscio} se $\wtM$ è una varietà differenziabile e $\pi : \wtM \longrightarrow M$ è una mappa liscia suriettiva tale che per ogni $p \in M$ esiste $U \subset M$ intorno aperto di $p$ tale che $\pi^{-1}(U)$ è unione disgiunta di aperti connessi $(V_i)_{i \in I}$ di $\wtM$ e $\pi|_{V_i} : V_i \longrightarrow U$ è un diffeomorfismo.
\end{defn}
\begin{thm}
    siano $M$ una varietà differenziabile liscia e $(\wtM, \pi)$ un rivestimento topologico di $M$, allora esiste unica struttura differenziabile su $\wtM$ tale che $\pi$ è rivestimento liscio.
\end{thm}
\begin{proof}
    consideriamo $\wtp \neq \wtq$ in $\wtM$, se $\pi(\wtp) = \pi(\wtq)$ e consideriamo l'intorno ben ricoperto $U$ di $\pi(\wtp) = \pi(\wtq)$. Necessariamente $\wtp$ e $\wtq$ non possono appartenere alla stessa componente connessa di $\pi^{-1}(U)$, dato che la restrizione di $\pi$ su di essa è un omeomorfismo. Siano $V_{i_0}, V_{i_1}$ le componenti connesse di $\pi^{-1}(U)$ che contengono $\wtp$ e $\wtq$ rispettivamente ($V_{i_0} \cap V_{i_1} = 0$). \\
    Se $\pi(\wtp) \neq \pi(\wtq)$ esistono $U_1, U_2$ intorni aperti di $\pi(\wtp)$ e $\pi(\wtq)$ rispettivamente tali che $U_1 \cap U_2 = \emptyset$. Poniamo $\wtU_1 := \pi^{-1}(U_1)$ e $\wtU_2 := \pi^{-1}(U_2)$, allora $\wtU_1 \cap \wtU_2 = \emptyset$. \\
    Dato $\wtp \in \wtM$ e $U$ intorno ben ricoperto di $\pi(\wtp)$, possiamo assumere che $U$ è una carta di $M$ (a meno di restringere), pertanto esiste $\vp$ omemomorfismo di $U$ in $\R^n$. Sia $V$ la componente connessa di $\pi^{-1}(U)$ che contiene $\wtp$, allora
    \begin{align}
        \wt{\vp} : V \longrightarrow \vp(U)
    \end{align}
    la mappa $\wt{\vp} : \wtq \in \wtM \mapsto \vp \circ \pi (\wtq)$ è un omeomorfismo. \\
    Facciamo vedere che i cambi di carta sono lisci, siano $(V, \wt{\vp})$ e $(W, \wt{\psi})$ due carte di $\wtM$, come sopra definita. Poniamoci nel caso interessante $V \cap W \neq \emptyset$, allora
    \begin{align}
        \wt{\psi} \circ \wt{\vp}^{-1} = (\psi \circ \pi|_W) \circ (\vp \circ \pi|_{V})^{-1} = \psi \circ \vp^{-1} 
    \end{align}
    che è liscia in $\vp(V \cap W)$.
\end{proof}

\chapter{Mappe proprie}
\begin{defn}
    siano $X, Y$ spazi topologici e $f : X \longrightarrow Y$, $f$ è detta \co{propria} se per ogni $K \subset Y$ compatto si ha $f^{-1}(K) \subset X$ compatto.
\end{defn}
\begin{lemma}
    se $X$ è uno spazio topologico compatto e $Y$ è $T_2$, ogni funzione continua $f : X \longrightarrow Y$ è propria.
\end{lemma}
\begin{proof}
    dato $K \subset Y$ compatto si ha $K$ chiuso, poiché $Y$ è $T_2$, pertanto $f^{-1}(K) \subset X$ è chiuso (dato che $f$ è continua) in un compatto, quindi è compatto.
\end{proof}
\begin{defn}
    sia $X$ uno spazio topologico, diciamo che $(x_n)_{n \in \N}$ \co{esce da ogni compatto di} $X$ se per ogni $K \subset X$ compatto esiste $N \in \N$ tale che $x_n \in X \setminus K$ \co{frequentemente} per ogni $n > N$.
\end{defn}
\begin{prop}
    siano $X, Y$ spazi topologici, $X$ secondo numerabile e $T_2$ e $f : X \longrightarrow Y$ continua, allora $f$ mappa propria se e solo se per ogni $(x_n)_{n \in N} \subset X$ che esce da ogni compatto si ha $(f(x_n))_{n \in \N}$ esce da ogni compatto.
\end{prop}
\begin{proof}
    $(\rightarrow)$ sia $(x_n)_{n \in \N} \subset X$ che esce da ogni compatto e consideriamo la successione in $Y$ data da $(f(x_n))_{n \in N}$. Supponiamo che $(f(x_n))_{n \in \N}$ è contenuta in un compatto $K$ di $Y$, allora $(x_n)_{n \in \N} \subset f^{-1}(K)$ che è un compatto in $X$ dato che $f$ è propria, che è assurdo. \\
    $(\leftarrow)$ sia $K \subset Y$ compatto e consideriamo l'insieme $f^{-1}(K) \subset X$. Per ipotesi $X$ è secondo numerabile e $T_2$, pertanto $f^{-1}(K)$ è compatto se e solo se è compatto per successioni, ovvero se e solo se ogni successione $(x_n)_{n \in \N} \subset f^{-1}(K)$ ammette sottosuccessione convergente in $f^{-1}(K)$. Supponiamo che $f^{-1}(K)$ non è compatto, allora $(x_n)_{n \in \N}$ non ammette sottosuccessione convergente in $Y$, ovvero esce da ogni compatto, quindi anche $(f(x_n))_{n \in \N}$ esce da ogni compatto, ma $(f(x_n))_{n \in \N} \subset K$, quindi si ottiene un assurdo.
\end{proof}
\begin{lemma}
    siano $M, N$ varietà topologiche e $f : M \longrightarrow N$ propria, allora $f$ è chiusa.
\end{lemma}
\begin{proof}
    sia $C \subset M$ chiuso e consideriamo $(y_n)_{n \in \N} \subset f(C)$ successione convergente in $N$, mostriamo che $y := \lim y_n \in C$. $N$ è localmente euclideo, pertanto esistono $U$ intorno aperto di $y$ e $\vp : U \longrightarrow \vp(U)$ omeomorfismo, inoltre esiste $K_y \subset \R^n$ intorno compatto di $\vp(y)$. Poniamo $K := \vp^{-1}(K_y)$, che è compatto dato che $\vp$ è un omeomorfismo, per definizione di convergenza in spazi topologici si ha che $y_n \in K$ definitivamente ($n \geq N$). Per ogni $n > N$ prendiamo $x_n \in f^{-1}(K \cap C)$ (una delle preimmagini di $y_n$). Allora esiste una sottosuccessione $(x_{n_k})_{k \in \N} \subset f^{-1}(K \cap C)$ convergente in $x \in f^{-1}(K \cap C)$, dato che $f^{-1}(K \cap C)$ è compatto ($f$ propria). Concludiamo che
    \begin{align}
        y := \lim_{n \rightarrow \infty} y_n = \lim_{k \rightarrow \infty} f(x_{n_k}) \in f(C)
    \end{align}
    quindi $C$ chiuso.
\end{proof}
\begin{prop}
    siano $M, N$ varietà differenziabili connesse e $f : M \longrightarrow N$ diffeomorfismo locale e propria. Allora $f$ è un rivestimento liscio (finito).
\end{prop}
\begin{proof}
    Notiamo immediatamente che se $f$ è un diffeomorfismo locale si ha $f$ aperta, chiusa. Segue che $f(M)$ è un aperto e chiuso in $N$, che è connesso, quindi $f(M) = N$, ovvero $f$ suriettiva. \\
    Dato $y \in N$, per ogni $x \in f^{-1}(y)$ esiste $U_x$ intorno aperto di $x$ in $M$ tale che $f|_{U_x} : U_x \longrightarrow f(U_x)$ è un diffeomorfismo. Inoltre, $f^{-1}(y)$ è un compatto ($f$ propria) discreto (in topologia di sottoinsieme), infatti i punti non possono accumularsi attorno a qualche altra preimmagine dato che $f$ è un diffeomorfismo locale. Segue che $f^{-1}(y)$ è un insieme finito (un compatto discreto è sempre finito). \\
    Costruiamo l'intorno ben ricoperto di $y$, quindi poniamo
    \begin{align}
        f^{-1}(y) = \{x_1, \dots, x_k\}
    \end{align}
    e
    \begin{align}
        V := \bigcap_{i = 1}^k f(U_{x_i}) \setminus f\left(M \setminus \bigcup_{i = 1}^k U_{x_i}\right)
    \end{align}
    La necessità di sottrarre l'insieme $f\left(M \setminus \bigcup_{i = 1}^k U_{x_i}\right)$ viene dal fatto che $f$ è un diffeomorfismo locale, non globale, pertanto, in generale, potrebbero esserci dei punti $x \in M \setminus \bigcup_{i = 1}^k U_{x_i}$ che vengono mappati in $f(U_{x_i})$ da $f$, per qualche $i = 1, \dots, k$, quindi, in tal caso, $f$ non sarebbe biettiva sulle componenti connesse di $f^{-1}(V)$ se non eliminassimo l'immagine tramite $f$ di $M \setminus \bigcup_{i = 1}^k U_{x_i}$ da $V$. \\
    Notiamo che $V$ è un aperto, dato che è la differenza tra un aperto e un chiuso. Se $V$ è sconnesso consideriamo la componente connessa di $V$ che contiene $y$. \\
    Mostriamo che $V$ è ben ricoperto. Poniamo $\wtU_{x_i} := f^{-1}(V) \cap U_{x_i}$ per ogni $i = 1, \dots, k$ e osserviamo che
    \begin{align}
        f^{-1}(V) = \bigcup_{i = 1}^k \wtU_{x_i}
    \end{align}
    Infatti se $x \in f^{-1}(V)$ si ha $f(x) \in V$, quindi $f(x) \in f(U_{x_i})$ per ogni $i = 1, \dots, k$ e $f(x) \not\in f\left(M \setminus \bigcup_{i = 1}^k U_{x_i}\right)$, quindi necessariamente $x \in U_{x_i}$ per qualche $i = 1, \dots, k$. Mentre, se $x \in \bigcup_{i = 1}^k \wtU_{x_i}$ necessariamente $x \in f^{-1}(V)$. \\
    Facciamo vedere che la restrizione suriettiva di $f$ su $\wtU_{x_i}$ è un diffeomorfismo per ogni $i = 1, \dots, k$. Osserviamo che è suriettiva per definizione di restrizione suriettiva, iniettiva dato che $\wtU_{x_i} \subset U_{x_i}$ e diffeomorfismo poiché restrizione di $f|_{U_{x_i}} : U_{x_i} \longrightarrow f(U_{x_i})$ che è un diffeomorfismo. 
\end{proof}

\chapter{Varietà complesse}
\begin{defn}
    siano $M$ una varietà topologica di dimensione $2n$ per qualche $n \geq 1$ e $(U, \vp)$, $(V, \psi)$ due carte di $M$
    \begin{align}
        \vp : U \longrightarrow \vp(U) \subset \cC^n \qquad \psi : V \longrightarrow \psi(V) \subset \cC^n
    \end{align}
    $(U, \vp)$ e $(V, \psi)$ sono dette \co{olomorficamente compatibili} se $U \cap V = \emptyset$ o se $U \cap V \neq \emptyset$ e
    \begin{align}
        \psi \circ \vp^{-1} : \vp(U \cap V) \longrightarrow \psi(U \cap V)
    \end{align}
    è olomorfa.
\end{defn}
\begin{defn}
    un \co{atlante olomorfo} di $M$ è una famiglia di carte olomorficamente compatibili di $M$. \\
    Analogamente al caso reale, la \co{struttura olomorfa} è la classe di equivalenza di atlanti equivalenti. \\ \\
    Fissata una struttura olomorfa, $M$ è detta \co{varietà complessa}.
\end{defn}
\begin{ex}
    gli aperti $\Omega \subset \cC^n$ sono varietà complesse con carte $(\Omega, \id_\Omega)$.
\end{ex}
\begin{defn}
    le varietà complesse di dimensione $1$ sono dette \co{superfici di Riemann}.
\end{defn}
\begin{ex}
    sia $\hat{\cC}$ la compattificazione ad un punto di $\cC$, ovvero $\hat{cC} = \cC \cup \{\infty\}$. Consideriamo le carte
    \begin{align}
        (\cC, \id_\cC) \qquad (\cC^* \cup \{\infty\}, \psi)
    \end{align}
    dove
    \begin{align}
        \psi : z \in \cC^* \mapsto \frac{1}{z}
    \end{align}
    con la convenzione $\psi(\infty) = 0$. Osserviamo che le due carte definite sono olomorficamente compatibili, infatti
    \begin{align}
        \psi \circ \id^{-1}_\cC : \cC^* \longrightarrow \cC^*
    \end{align}
    è olomorfa, dato che $z \mapsto 1/z$ è olomorfa in $\cC^*$.
\end{ex}

\chapter{Partizioni dell'unità}
\begin{defn}
    sia $X$ uno spazio topologico, un'\co{esaustione normale di compatti} è una famiglia di compatti $(K_i)_{i \in \N}$ tale che: \\
    \gr{1.} $\bigcup_{i \in \N} K_i = X$. \\
    \gr{2.} $K_i \subset \mathring{K}_{i + 1}$ per ogni $i \in \N$.
\end{defn}
\begin{ex}
    la famiglia $(K_i)_{i \in \N} \subset \R^n$ dove $K_i := \overline{D(0, i)}$ è un'esaustione normale di compatti di $\R^n$.
\end{ex}
\begin{defn}
    siano $X$ uno spazio topologico e $(U_\alpha)_\alpha$ un ricoprimento di $X$. Un \co{raffinemanto} di $(U_\alpha)_\alpha$ è un ricoprimento di $X$ $(V_\beta)_\beta$ tale che per ogni $\beta$ esiste $\alpha$ tale che $V_\beta \subset U_\alpha$.
\end{defn}
\begin{defn}
    siano $X$ uno spazio topologico e $(U_\alpha)_\alpha$ ricoprimento aperto di $X$. $(U_\alpha)_\alpha$ è detto \co{localmente finito} se per ogni $p \in X$ esiste $U$ intorno di $p$ che interseca con un numero finito di elementi del ricoprimento $(U_\alpha)_\alpha$.
\end{defn}
\begin{ex}
    sia $X = \R$ e consideriamo il ricoprimento aperto $\U = (U_n)_{n \in \N}$, dove $U_n := (-\infty, n)$ per ogni $n \in \N$. Osserviamo che $\U$ non è localmente finito, infatti ogni $x \in \R$ non possiede nessun intorno che interseca con un numero finito di elementi di $\U$. 
\end{ex}
\begin{defn}
    uno spazio topologico $X$ è detto \co{paracompatto} se ogni ricoprimento aperto di $X$ ammette raffinamento localmente finito.
\end{defn}
\begin{rmk}
    Osserviamo che se $X$ è uno spazio topologico localmente euclideo, allora $X$ è anche localmente compatto. Infatti, dato $x \in X$ esiste $U$ intorno aperto di $x$ in $X$ e $\vp : U \longrightarrow \vp(U) \subset \R^n$ omeomorfismo, quindi, fissato $0 < \varepsilon < \inf \{\|\vp(x) - z\| : z \in \R^n \setminus \vp(U)\}$, consideriamo
    \begin{align}
        V_\varepsilon := \overline{D(\vp(x), \varepsilon)} := \{y \in \R^n : \|\vp(x) - y\| \leq n\}
    \end{align}
    Allora $\vp^{-1}(V_\varepsilon) \subset U$ è un intorno compatto di $x$ ($\vp$ omeomorfismo).
\end{rmk}
\begin{thm}
    sia $X$ uno spazio topologico localmente compatto e $T_2$, se $X$ è secondo numerabile allora $X$ ammette un'esaustione normale di compatti, è paracompatto ed è metrizzabile.
\end{thm}
\begin{proof}
    sia $X$ secondo numerabile e consideriamo una base numerabile $\B = (B_i)_{i \in \N}$ della topologia di $X$. Osserviamo che possiamo supporre che $\B$ sia composta da intorni relativamente compatti, dato che $X$ è localmente compatto. \\
    Per ogni $n \in \N$ poniamo
    \begin{align}
        C_n := \overline{\bigcup_{i = 1}^n B_i}
    \end{align}
    Osserviamo che la famiglia $(C_i)_{i \in \N}$ è un'esaustione di compatti, infatti
    \begin{align}
        C_i \subset C_{i + 1} \qquad X = \bigcup_{i \in \N} C_i
    \end{align}
    Tuttavia non è detto che sia normale, ovvero non è detto che $C_i \subset \mathring{C}_{i + 1}$. \\
    Ragioniamo per induzione, fissiamo $K_1 := \overline{B}_1$, supponiamo di aver definito anche $K_n$ per qualche $n > 1$ e costruiamo $K_{n + 1}$ compatto tale che $K_n \subset \mathring{K}_{n + 1}$. Per ipotesi $K_n$ è compatto, pertanto esistono finiti $x_1, \dots, x_k \in K_n$ e loro intorno aperti relativamente compatti $V_{x_1}, \dots, V_{x_k} \subset X$ tali che
    \begin{align}
        K_n \subset V_{x_1} \cup \dots \cup V_{x_k}
    \end{align}
    Allora poniamo
    \begin{align}
        K_{n + 1} := \overline{\bigcup_{i = 1}^{n + 1} B_i} \cup \overline{\bigcup_{j = 1}^k V_{x_j}}
    \end{align}
    Si vede immediatamente che $K_{n + 1}$ è compatto dato che è unione finita di chiusure di aperti relativemente compatti in un $T_2$. Inoltre, $K_n \subset \mathring{K}_{n + 1}$, infatti contiene $V_{x_1}, \dots, V_{x_k}$ e anche $B_1, \dots, B_{n + 1}$. Infine, si vede immediatamente che
    \begin{align}
        \bigcup_{i \in \N} K_i = X
    \end{align}
    A questo punto consideriamo un ricoprimento aperto $\U = (U_\alpha)_\alpha$ di $X$ e costruiamo un raffinamento localmente finito di $\U$. \\
    Per ogni $j \in \N$ consideriamo l'insieme
    \begin{align}
        A_j := \mathring{K}_{j + 1} \setminus K_{j - 1}
    \end{align}
    con la convenzione $K_{-1} = \emptyset$. Osserviamo che $A_j$ è un aperto per ogni $j \in \N$, dato che $K_{j - 1} \subset \mathring{K}_{j + 1}$, inoltre, è relativamente compatto, infatti la sua chiusura è contenuta in $K_{j + 1}$, pertanto esiste un numero finito di elementi $U^j_{\alpha_1}, \dots, U^j_{\alpha_k}$ tali che
    \begin{align}
        A_j \subset U^j_{\alpha_1} \cup \dots \cup U^j_{\alpha_{k_j}}
    \end{align}
    Consideriamo ora gli aperti $W^j_{\alpha_i} := A_j \cap U^j_{\alpha_i}$ per ogni $j \in \N$ e ogni $i = 1, \dots, k_j$. Gli aperti $\W = \{W^j_{\alpha_i} : j \in \N, i = 1, \dots, k_j\}$ formano un ricoprimento aperto numerabile di $X$ (dato che si tratta di unione numerabile di famiglie finite di aperti). Inoltre, per ogni $x \in X$ esiste $j \in \N$ tale che $x \in A_j$ e quindi esiste un intorno aperto $V$ di $x$ tale che $V \subset A_j$. Segue che $V$ interseca solamente con un numero finito di elementi di $\W$, quindi $\W$ raffinamento localmente finito di $\U$, ovvero $X$ paracompatto. \\
    Dimostriamo ora che $X$ è metrizzabile, ovvero costruiamo una funzione distanza $d : X \longrightarrow [0, +\infty)$ (da fare).
\end{proof}
\begin{defn}
    sia $M$ una varietà liscia, una \co{partizione dell'unità} è una famiglia di funzioni $(\rho_\alpha)_\alpha \subset \C^\infty(M)$ tale che: \\
    \gr{1.} $0 \leq \rho_\alpha \leq 1$ per ogni $\alpha$. \\
    \gr{2.} $\{\operatorname{supp} \rho_\alpha\}_\alpha$ è un ricoprimento localmente finito di $M$. \\
    \gr{3.} $\sum_\alpha \rho_\alpha = 1$. \\
    Diremo che $(\rho_\alpha)_\alpha$ è \co{subordinata} a un ricoprimento aperto $(U_\alpha)_\alpha$ se $\{\operatorname{supp}_\alpha\}_\alpha$ è un raffinamento di $(U_\alpha)_\alpha$, in particolare se $\operatorname{supp}_\alpha \subset U_\alpha$ per ogni $\alpha$.
\end{defn}
\begin{rmk}
    per la proprietà $(2)$, per ogni $x \in M$ esiste $U$ intorno aperto di $x$ tale che $U$ interseca solamente con un numero finito di elementi di $\{\operatorname{supp}_\alpha\}_\alpha$, pertanto la serie definita in $(3)$ è in realtà una somma finita (puntualmente). 
\end{rmk}

\end{thm}

        \end{cases}
    \end{align}

    \gr{3.} $h(t) = 0$ per $t \geq 2$, \\
    detta funzione \co{cut-off}.
\end{proof}
