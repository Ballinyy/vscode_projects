\chapter{Misure su $\sigma$-algebre}
\section{Algebre e $\sigma$-algebre}
\begin{defn}
sia $X$ un insieme, $\mathcal{A} \subset \mathcal{P}(X)$ è detta \textit{algebra} se: \\
\textbf{1.} $\emptyset \in \mathcal{A}$. \\
\textbf{2.} $A \cup B \in \mathcal{A}$ per ogni $A, B \in \mathcal{A}$ (chiusa rispetto a unioni finite). \\
\textbf{3.} $X \setminus A \in \mathcal{A}$ per ogni $A \in \mathcal{A}$ (chiusu rispetto al passaggio al complementare).
\end{defn}
\begin{rmk}
se $A, B \in \mathcal{A}$ anche $A \cap B \in \mathcal{A}$, infatti $X \setminus A, X \setminus B \in \mathcal{A}$, per ipotesi, e
\begin{align}
    (X \setminus A) \cup (X \setminus B) = X \setminus (A \cap B)
\end{align}
per il \textit{teorema di De Morgan}, quindi anche $X \setminus (X \setminus (A \cap B)) = A \cap B \in \mathcal{A}$. Concludiamo che un'algebra è anche chiusa per intersezioni finite.
\end{rmk}
\begin{defn}
$\mathcal{A} \subset \mathcal{P}(X)$ è detta $\sigma$-\textit{algebra} se $\mathcal{A}$ è un'algebra e data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ si ha $A = \bigcup_{n \in \mathbb{N}} A_n \in \mathcal{A}$, ovvero se è chiusa anche per unioni numeribili.
\end{defn}
\begin{rmk}
sempre per il \textit{teorema di De Morgan}, se $\mathcal{A}$ è una $\sigma$-algebra è anche chiusa per intersezioni numerabili.
\end{rmk}
\begin{ex}
$(1)$ $\mathcal{P}(X)$ e $\{\emptyset, X\}$ sono esempi banali di $\sigma$-algebre. \\ \\
$(2)$ Dato $X$ insieme tale che $|X| = \infty$ si ha che
\begin{align}
    \mathcal{A} = \{E \in \mathcal{P}(X) : |E| < \infty \text{ o } |X \setminus E| < \infty|\}
\end{align}
è un'algebra. Infatti, dati $E_1, E_2 \in \mathcal{A}$ tali che $|E_1|, |E_2| < \infty$ si ha
\begin{align}
    |E_1 \cup E_2| = |E_1| + |E_2| - |E_1 \cap E_2| < \infty
\end{align}
Invece, se $|X \setminus E_1|, |X \setminus E_2| < \infty$ si ha
\begin{align}
    |X \setminus (E_1 \cup E_2)| = |(X \setminus E_1) \cap (X \setminus E_2)| < \infty
\end{align}
ovvero $X \setminus (E_1 \cup E_2) \in \mathcal{A}$, quindi, per definizione di $\mathcal{A}$, anche $E_1 \cup E_2 \in \mathcal{A}$. \\
Tuttavia, $\mathcal{A}$ non è una $\sigma$-algebra, infatti se $(E_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ non è detto che $E = \bigcup_{n \in \mathbb{N}} E_n$ sia finito o che $X \setminus E$ lo sia (nel primo caso è sufficiente prendere una successione di insieme disgiunti, anche finiti, nel secondo, se $X \cong \mathbb{N}$, è sufficiente porre $E_k = \{k^2\}$, $k \in \mathbb{N}$, lo stesso se $X$ è non numerabile) \\ \\
$(3)$ Dato $X$ insieme tale che $|X| = \infty$ si ha che
\begin{align}
    \mathcal{A} = \{E \in \mathcal{P}(X) : E \text{ o } X \setminus E \text{ è numerabile}\}
\end{align}
è una $\sigma$-algebra. Infatti, data $(E_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ tale che $E_k$ è numerabile per ogni $k \in \mathbb{N}$ si ha
\begin{align}
    E = \bigcup_{n \in \mathbb{N}} E_n
\end{align}
è numerabile. Mentre se $X \setminus E_k$ è numerabile per ogni $k \in \mathbb{N}$ si ha
\begin{align}
    X \setminus \bigcup_{n \in \mathbb{N}} E_n = \bigcap_{n \in \mathbb{N}} (X \setminus E_n)
\end{align}
che è numerabile poiché contenuto in $X \setminus E_k$ per ogni $k \in \mathbb{N}$, quindi anche $E \in \mathcal{A}$. \\ \\
$(4)$ Posto $X = [0, 1)$ e
\begin{align}
    \mathcal{A} = \emptyset \cup \left\{\bigcup_{i = 1}^n [a_i, b_i) : 0 \leq a_1 < b_1 \leq a_2 < b_2 \leq \dots \leq a_n < b_n \leq 1, n \in \mathbb{N}\right\}
\end{align}
si ha che $\mathcal{A}$ è un'algebra. Infatti, dati $[a, b), [c, d) \in \mathcal{A}$, $0 < a < b < 1, 0 < c < d < 1$, si ha
\begin{align}
    [0, 1) \setminus [a, b) = [0, a) \cup [b, 1) \in \mathcal{A}
\end{align}
per definizione di $\mathcal{A}$, se $a = 0$ e/o $d = 1$ uno dei due insiemi diventa l'insieme vuoto, mentre se $a = 0$ e $b = 1$ si ottiene il vuoto, che è contenuto in $\mathcal{A}$. \\
Mentre
\begin{align}
    [a, b) \cup [c, d) = \begin{cases}
        [a, d) \qquad &a < c \leq b < d \\
        [a, b) \cup [c, d) \qquad &a < b \leq c < d \\
        [c, b) \qquad & c < a \leq d < b \\
        [a, b) \qquad &a \leq c < d \leq b \\
        [c, d) \qquad &c \leq a < b \leq d
    \end{cases}
\end{align}
In ogni caso appartiene ad $\mathcal{A}$ per definizione. \\
Tuttavia, $\mathcal{A}$ non è una $\sigma$-algebra, infatti, posto
\begin{align}
    I_k = \left(\frac{1}{4k + 1}, \frac{1}{4k}\right) \qquad k \in \mathbb{N}
\end{align}
si ha $I_k \in \mathcal{A}$ per ogni $k \in \mathbb{N}$ ma
\begin{align}
    \bigcup_{k \in \mathbb{N}} I_k
\end{align}
è unione numerabile disgiunta, pertanto non può essere espresso come unione finita di intervalli della forma $[a, b)$, $a < b$.
\end{ex}
\begin{defn}
sia $X$ un insieme e $\mathcal{E} \subset \mathcal{P}(X)$ un'algebra, chiamiamo $\sigma$-\textit{algebra generata da} $\mathcal{E}$ la più piccola $\sigma$-algebra che contiene $\mathcal{E}$ è la denotiamo con $\sigma(\mathcal{E})$.
\end{defn}
\begin{ex}
$(1)$ poniamo $X = \{1, \dots, n\}$ e
\begin{align}
    \mathcal{E} = \{\{1\}, \{2\}, \dots, \{n - 1\}, \{n - 1, n\}\}
\end{align}
La $\sigma$-algebra generata da $\mathcal{E}$ deve contenere $\mathcal{P}(\{1, \dots, n - 2\})$ e $\{n - 1, n\}$, pertanto
\begin{align}
    \sigma(\mathcal{E}) = \{A \cup B : A \in \mathcal{P}(\{1, \dots, n - 1\}), B = \emptyset, \{n - 1, n\}\}
\end{align}
$(2)$ Posto $X = \mathbb{R}$ e
\begin{align}
    \mathcal{E} = \{[a, b) : a < b, a, b \in \mathbb{R}\}
\end{align}
si ha che $\sigma(\mathcal{E}) = \mathcal{B}(\mathbb{R})$, dove $\mathcal{B}(\mathbb{R})$ è la $\sigma$-algebra dei \textit{boreliani} di $\mathbb{R}$, ovvero la $\sigma$-algebra generata dagli aperti di $\mathbb{R}$. \\
Per dimostrare l'uguaglianza è sufficiente dimostrare che i generatore di un insieme si trovano nella $\sigma$-algebra generata dall'altro e viceversa. In particolare, posti $a < b, c < d$ e scelto $\alpha$ opportuno affinché $c + \alpha < d$, si ha
\begin{align}
    [a, b) &= \bigcap_{n \in \mathbb{N}} \left(a - \frac{1}{n}, b\right) \\
    (c, d) &= \bigcup_{n \in \mathbb{N}} \left[c + \frac{\alpha}{n}, d\right)
\end{align}
\end{ex}

\section{Misure}
\begin{defn}
dato $X$ insieme e $\mathcal{A} \subset \mathcal{P}(X)$ una $\sigma$-algebra, $\mu : \mathcal{A} \longrightarrow [0, \infty]$ è detta \textit{misura} se: \\
\textbf{1.} data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ tale che $A_i \cap A_j = \emptyset$ se $i \neq j$ si ha
\begin{align}
    \mu\left(\bigcup_{n \in \mathbb{N}} A_n\right) = \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
ovvero se è $\sigma$-\textit{additiva}. \\
\textbf{2.} $\mu(\emptyset) = 0$.
\end{defn}
\begin{rmk} si vede immediatamente che ogni misura è anche una funzione additiva, infatti
\begin{align}
    \mu(A \cup B) = \mu(A \cup B \cup \emptyset \cup \emptyset \cup \dots) = \mu(A) + \mu(B)
\end{align}
\end{rmk}
\begin{ex}
$(1)$ dato $X$ insieme discreto e $\mathcal{A} = \mathcal{P}(X)$, poniamo
\begin{align}
    \mu : A \in \mathcal{P}(X) \mapsto \begin{cases}
        \infty \qquad &|A| = \infty \\
        |A| \qquad &|A| < \infty
    \end{cases}
\end{align}
Inoltre, data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{P}(X)$, poniamo
\begin{align}
    \mu\left(\bigcup_{n \in \mathbb{N}} A_n\right) = \begin{cases}
        \infty \qquad &|A_i| = \infty \text{ per qualche } i \in \mathbb{N} \\
        \sum_{n \in \mathbb{N}} |A_n| \qquad &A_i \cap A_j = \emptyset, i \neq j
    \end{cases}
\end{align}
Si vede immediatamente che $\mu$ è $\sigma$-additiva e che $\mu(\emptyset) = 0$ (per defizione), quindi $\mu$ è una misura, in particolare è detta \textit{misura di conteggio}. \\ \\
$(2)$ Poniamo $X = \mathbb{N}$ e $\mathcal{A} = \mathcal{P}(X)$, allora
\begin{align}
    \mu : A \in \mathcal{P}(X) \mapsto \begin{cases}
        \sum_{a \in A} 2^{-a} \qquad &|A| < \infty \\
        \infty \qquad &|A| = \infty
    \end{cases}
\end{align}
è additiva, ma non $\sigma$-additiva. Infatti, dati $A, B \in \mathcal{P}(X)$ disgiunti, si ha che
\begin{align}
    \mu(A \cup B) = \sum_{a \in A}2^{-a} + \sum_{b \in B}2^{-b} = \mu(A) + \mu(B)
\end{align}
Mentre, posto $A_k = \{k\}$ per ogni $k \in \mathbb{N}$, si ha
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n = \mathbb{N}
\end{align}
Pertanto
\begin{align}
    \mu(A) = \infty \neq \sum_{n \in \mathbb{N}} 2^{-n} = \sum_{n \in \mathbb{N}} \mu(A)
\end{align}
\end{ex}
\begin{rmk}\label{Monotone}
dati $\mu$ additiva su $\mathcal{A}$ algebra e $A, B \in \mathcal{A}$ tali che $A \subset B$, si ha
\begin{align}
    \mu(A) \leq \mu(B) \qquad \mu(B \setminus A) = \mu(B) - \mu(A)
\end{align}
Infatti $B = A \cup (B \setminus A)$, unione disgiunta, quindi
\begin{align}
    \mu(B) = \mu(A) + \mu(B \setminus A) \geq \mu(A)
\end{align}
per positività di $\mu$.
\end{rmk}
\begin{rmk} dati $\mu$ additiva su $\mathcal{A}$ $\sigma$-algebra e $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ disgiunti, si ha che
\begin{align}
    \mu\left(\bigcup_{n \in \mathbb{N}} A_n\right) \geq \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
ovvero $\mu$ $\sigma$-\textit{supadditiva}. Infatti
\begin{align}
    \bigcup_{n \leq k} A_n \subset \bigcup_{n \in \mathbb{N}} A_n
\end{align}
per ogni $k \in \mathbb{N}$, pertanto, per l'osservazione \ref{Monotonia}, si ha
\begin{align}
    \mu\left(\bigcup_{n \leq k} A_n\right) \geq \mu\left(\bigcup_{n \leq k} A_n\right) = \sum_{n \leq k} \mu(A_n)
\end{align}
per ogni $k \in \mathbb{N}$, passando al limite
\begin{align}
    \mu\left(\bigcup_{n \in \mathbb{N}} A_n\right) \geq \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
\end{rmk}
\begin{defn}
dati $X$ insieme e $\mathcal{A} \subset \mathcal{P}(X)$ $\sigma$-algebra, $\mu : \mathcal{A} \longrightarrow [0, \infty]$ è detta $\sigma$-\textit{subadditiva} se, dato $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$, si ha
\begin{align}
    \mu\left(\bigcup_{n \in \mathbb{N}} A_n\right) \leq \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
\end{defn}
\begin{prop}\label{Add->SubAdd}
dati $X$ insieme e $\mathcal{A} \subset \mathcal{P}(X)$ $\sigma$-algebra, se $\mu : \mathcal{A} \longrightarrow [0, \infty]$ è $\sigma$-additiva allora $\mu$ è anche $\sigma$-subadditiva.
\end{prop}

\begin{proof}
dato $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$, poniamo
\begin{align}
    \widetilde{A}_n = A_n \setminus \bigcup_{k \leq n - 1} A_k
\end{align}
Allora $\widetilde{A}_i \cap \widetilde{A}_j = \emptyset$ per ogni $i \neq j$ e
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n = \bigcup_{n \in \mathbb{N}} \widetilde{A}_n
\end{align}
Pertanto
\begin{align}
    \mu(A) = \mu\left(\bigcup_{n \in \mathbb{N}} \widetilde{A}_n\right) = \sum_{n \in \mathbb{N}} \mu(\widetilde{A}_n) \leq \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
dato che $\widetilde{A}_k \subset A_k$ per ogni $k \in \mathbb{N}$.
\end{proof}
\begin{cor}
dati $X$ insieme, $\mathcal{A} \subset \mathcal{P}(X)$ $\sigma$-algebra e $\mu : \mathcal{A} \longrightarrow [0, \infty]$ additiva, sono equivalenti: \\
\textbf{1.} $\mu$ $\sigma$-additiva. \\
\textbf{2.} $\mu$ $\sigma$-subadditiva.
\end{cor}

\begin{proof}
$(1 \rightarrow 2)$ per la proposizione \ref{Add->SubAdd}. \\
$(2 \rightarrow 1)$ Abbiamo visto che se $\mu$ è additiva su $\mathcal{A}$ allora $\mu$ è anche $\sigma$-supadditiva, pertanto la tesi segue confrontando con l'ipotesi.
\end{proof}
\begin{defn}
siano $X$ un insieme e $(A_i)_{i \in I} \subset \mathcal{P}(X)$, con $I$ insieme di indici, diremo che $(A_i)_{i \in I}$ è \textit{crescente} (\textit{decrescente}) se $A_i \subset A_j$ ($A_i \sup A_j$) per ogni $i > j$.
\end{defn}
\begin{prop}\label{LimCresc}
dati $X$ insieme, $\mathcal{A} \subset \mathcal{P}(X)$ algebra, $\mu$ $\sigma$-additiva e $(A_n)_{n \in \mathbb{N}}$ crescente, posto
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n \in \mathcal{A}
\end{align}
si ha $\mu(A) = \lim_n \mu(A_n)$.
\end{prop}

\begin{proof}
possiamo scrivere
\begin{align}
    A = A_1 \cup \bigcup_{n \in \mathbb{N}} A_{n + 1} \setminus A_n
\end{align}
unione disgiunta, inoltre sappiamo che $\mu(A_{k + 1} \setminus A_k) = \mu(A_{k + 1}) - \mu(A_k)$ per ogni $k \in \mathbb{N}$. Pertanto
\begin{align}
    \mu(A) = \mu(A_1) + \sum_{n \in \mathbb{N}} \mu(A_{n + 1}) - \mu(A_n) = \lim_n \mu(A_n)
\end{align}
dato che $\mu(A_k) = \mu(A_1) + \sum_{n \leq k - 1} \mu(A_{n + 1}) - \mu(A_n)$ per ogni $k \in \mathbb{N}$.
\end{proof}
\begin{prop}\label{LimDecresc}
dati $X$ insieme, $\mathcal{A} \subset \mathcal{P}(X)$ algebra, $\mu$ $\sigma$-additiva tale che $\mu(X) < \infty$ e $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ decrescente, posto
\begin{align}
    A := \bigcap_{n \in \mathbb{N}} A_n
\end{align}
si ha $\mu(A) = \lim_n \mu(A_n)$.
\end{prop}

\begin{proof} poniamo
\begin{align}
    B_n := X \setminus A_n
\end{align}
per ogni $n \in \mathbb{N}$, allora
\begin{align}
    X \setminus A = \bigcup_{n \in \mathbb{N}} B_n
\end{align}
Notiamo che $(B_n)_{n \in \mathbb{N}}$ è crescente, quindi possiamo scrivere
\begin{align}
    X \setminus A = B_1 \cup \bigcup_{n \in \mathbb{N}} B_{n + 1} \setminus B_n
\end{align}
unione disgiunta, inoltre sappiamo che $\mu(B_{k + 1} \setminus B_k) = \mu(B_{k + 1}) - \mu(B_k)$ per ogni $k \in \mathbb{N}$. Pertanto
\begin{align}
    \mu(X \setminus A) = \mu(B_1) + \sum_{n \in \mathbb{N}} \mu(B_{n + 1}) - \mu(B_n) = \lim_n \mu(B_n)
\end{align}
In particolare $\mu(X \setminus A) = \mu(X) - \mu(A)$, mentre $\mu(B_n) = \mu(X) - \mu(A_n)$, quindi si ottiene la tesi.
\end{proof}
\begin{cor}
se $\mu(A) = \lim_n \mu(A_n)$ e $\mu$ è additiva, si ha $\mu$ $\sigma$-additiva.
\end{cor}

\begin{proof}
data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ famiglia disgiunta, per ogni $k \in \mathbb{N}$ poniamo
\begin{align}
    \widetilde{A}_n = \bigcup_{k \leq n} A_k
\end{align}
Allora $(\widetilde{A}_n)_{n \in \mathbb{N}}$ è crescente e 
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n = \bigcup_{n \in \mathbb{N}} \widetilde{A}_n
\end{align}
Pertanto, per la proposizione \ref{LimCresc}, si ha che
\begin{align}
    \mu(A) = \lim_n \mu(\widetilde{A}_n) = \lim_n \sum_{k \leq n} \mu(A_k) = \sum_{n \in \mathbb{N}} \mu(A_n)
\end{align}
quindi la tesi.
\end{proof}
\begin{defn} siano $X$ un insieme e $(A_n)_{n \in \mathbb{N}} \subset \mathcal{P}(X)$, poniamo
\begin{align}
    \limsup_n A_n := \bigcap_{n \in \mathbb{N}}\left(\bigcup_{k \geq n} A_k\right) \qquad \liminf_n A_n := \bigcup_{n \in \mathbb{N}}\left(\bigcap_{k \geq n} A_k\right) 
\end{align}
\end{defn}
\begin{rmk}
$\bigcup_{k \geq n} A_k$ è decrescente in $n$, mentre $\bigcap_{k \geq n} A_k$ è crescente in $n$. \\
Se $(A_n)_{n \in \mathbb{N}}$ è crescente si ha
\begin{align}
    \bigcup_{k \geq n_1} A_k = \bigcup_{k \geq n_2} A_k
\end{align}
per ogni $n_2 \geq n_1$, pertanto
\begin{align}
    \limsup_n A_n = \lim_n A_n := \bigcup_{n \in \mathbb{N}} A_n
\end{align}
Mentre, se $(A_n)_{n \in \mathbb{N}}$ è decrescente si ha
\begin{align}
    \bigcap_{k \geq n_1} A_k = \bigcap_{k \geq n_2} A_k
\end{align}
per ogni $n_2 \geq n_1$, pertanto
\begin{align}
    \liminf_n A_n = \lim_n A_n := \bigcap_{n \in \mathbb{N}} A_n
\end{align}
\end{rmk}
\begin{lemma}
dati $X$ insieme, $\mathcal{A} \subset \mathcal{P}(X)$ $\sigma$-algebra, $\mu : \mathcal{A} \longrightarrow [0, \infty)$ $\sigma$-additiva e finita e $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$, allora
\begin{align}
    \mu(\liminf_n A_n) \leq \liminf_n \mu(A_n) \leq \limsup_n \mu(A_n) \leq \mu(\limsup_n A_n)
\end{align}
\end{lemma}

\begin{proof}
per definizione
\begin{align}
    \liminf_n A_n &= \bigcup_{n \in \mathbb{N}}\left(\bigcap_{k \geq n} A_k\right) \qquad \limsup_n A_n = \bigcap_{n \in \mathbb{N}}\left(\bigcup_{k \geq n} A_k\right) \\
    \liminf_n \mu(A_n) &= \sup_{n \in \mathbb{N}} \inf_{k \geq n} \mu(A_k) \qquad \limsup_n \mu(A_n) = \inf_{n \in \mathbb{N}} \sup_{k \geq n}\mu(A_k)
\end{align}
Poniamo
\begin{align}
    B_n := \bigcap_{k \geq n} A_k
\end{align}
per ogni $n \in \mathbb{N}$. Notiamo che $(B_n)_{n \in \mathbb{N}}$ è crescente, per la proposizione \ref{LimCresc} si ha
\begin{align}
    \mu(\liminf_n A_n) = \lim_n \mu(B_n) \leq \lim_n \inf_{k \geq n} \mu(A_k) \leq \sup_{n \in \mathbb{N}} \inf_{k \geq n} \mu(A_k) = \liminf_n \mu(A_n)
\end{align}
Analogamente, poniamo
\begin{align}
    C_n := \bigcup_{k \geq n} A_k
\end{align}
per ogni $n \in \mathbb{N}$. Notiamo che $(C_n)_{n \in \mathbb{N}}$ è decrescente, per la proposizione \ref{LimDecresc} si ha
\begin{align}
    \mu(\limsup_n A_n) = \lim_n \mu(C_n) \geq \lim_n \sup_{k \geq n} \mu(A_k) \geq \inf_{n \in \mathbb{N}} \sup_{k \geq n} \mu(A_k) = \limsup_n \mu(A_n)
\end{align}
quindi la tesi.
\end{proof}
\begin{lemma}[di Borel-Cantelli]
dati $X$ insieme, $\mathcal{A} \subset \mathcal{P}(X)$ $\sigma$-algebra, $\mu : \mathcal{A} \longrightarrow [0, \infty)$ $\sigma$-additiva e finita e $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ tale che
\begin{align}
    \sum_{n \in \mathbb{N}} \mu(A_n) < \infty
\end{align}
si ha $\mu(\limsup_n A_n) = 0$.
\end{lemma}

\begin{proof}
poniamo
\begin{align}
    B_n := \bigcup_{k \geq n} A_k
\end{align}
per ogni $n \in \mathbb{N}$. $(B_n)_{n \in \mathbb{N}}$ è decrescente, per la proposizione \ref{LimDecresc} si ha
\begin{align}
    \mu(\limsup_n A_n) = \lim_n \mu(B_n) \leq \lim_n \sum_{k \geq n} \mu(A_k) = 0
\end{align}
dato che per ipotesi $\sum_{n \in \mathbb{N}} \mu(A_n) < \infty$.
\end{proof}
A questo punto, data una funzione additiva $\mu$ definita su un'algebra $\mathcal{A}$, siamo interessati a costruire un'estensione $\overline{\mu}$ definita su $\sigma(\mathcal{A})$ che sia $\sigma$-additiva e quindi una misura. \\
Ad esempio, posta
\begin{align}
    \mathcal{A} = \emptyset \cup \left\{\bigcup_{i = 1}^n [a_i, b_i) : 0 \leq a_1 < b_1 \leq \dots < b_n \leq 1\right\}
\end{align}
vogliamo estendere la misura naturale
\begin{align}
    \mu : \bigcup_{j = 1}^s [c_j, d_j) \in \mathcal{A} \mapsto \sum_{j = 1}^s d_j - c_j
\end{align}
Può capitare che, data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{A}$ (algebra arbitraria), si abbia
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n \in \mathcal{A}
\end{align}
Pertanto, poiché deve valere $\overline{\mu}|_{\mathcal{A}} = \mu$, richiediamo che $\mu$ sia anche $\sigma$-additiva (anche se $\mathcal{A}$ è solo un'algebra).
\begin{defn}
dato $X$ insieme e $\mathcal{M} \subset \mathcal{P}(X)$, $\mathcal{M}$ è detta \textit{classe monotona} se per ogni $(A_n)_{n \in \mathbb{N}} \subset \mathcal{M}$ monotona si ha $\lim_n A_n \in \mathcal{M}$.
\end{defn}
\begin{rmk}
se $\mathcal{M}$ è un'algebra e classe monotona si ha $\mathcal{M}$ $\sigma$-algebra, infatti data $(A_n)_{n \in \mathbb{N}} \subset \mathcal{M}$ poniamo
\begin{align}
    B_n := \bigcup_{k \leq n} A_k
\end{align}
Allora $(B_n)_{n \in \mathbb{N}} \subset \mathcal{M}$ è crescente, pertanto
\begin{align}
    A := \bigcup_{n \in \mathbb{N}} A_n = \bigcup_{n \in \mathbb{N}} B_n \in \mathcal{M}
\end{align}
\end{rmk}
\begin{thm}[di Halmos]\label{ThmHalm}
dati $X$ insieme $\mathcal{A}, \mathcal{M} \subset \mathcal{P}(X)$ con $\mathcal{A}$ algebra e $\mathcal{M}$ classe monotona tali che $\mathcal{A} \subset \mathcal{M}$, allora $\sigma(\mathcal{A}) \subset \mathcal{M}$.
\end{thm}

\begin{proof}
sia $\mathcal{M}_0$ la classe monotona minimale che contiene $\mathcal{A}$ (esiste per il \textit{lemma di Zorn}). Dato $A \in \mathcal{A}$ poniamo
\begin{align}
    \mathcal{M}_A = \{B \in \mathcal{M}_0 : A \cup B, A \cap B, A \setminus B, B \setminus A \in \mathcal{M}_0\} 
\end{align}
Notiamo che $\mathcal{M}_A$ è una classe monotona, infatti, data $(B_n)_{n \in \mathbb{N}} \subset \mathcal{M}_A$ monotona, si ha
\begin{align}
    B := \lim_n B_n \in \mathcal{M}_0
\end{align}
Inoltre $(A \cup B_n)_n, (A \cap B_n)_n, (A \setminus B_n)_n$ e $(B_n \setminus A)_n$ sono tutte successioni monotone in $\mathcal{M}_0$, pertanto il loro limite appartiene a $\mathcal{M}_0$ e quindi, per definizione, anche a $\mathcal{M}_A$.
\end{proof}
